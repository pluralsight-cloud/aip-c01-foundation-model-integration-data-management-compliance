# Create the ConversationHistory table
aws dynamodb create-table \
  --table-name ConversationHistory \
  --attribute-definitions AttributeName=SessionId,AttributeType=S \
  --key-schema AttributeName=SessionId,KeyType=HASH \
  --billing-mode PAY_PER_REQUEST

# Enable TTL so conversation history automatically expires after 24 hours
aws dynamodb update-time-to-live \
  --table-name ConversationHistory \
  --time-to-live-specification "Enabled=true, AttributeName=ExpiresAt"

# Inspect stored conversation history
aws dynamodb scan --table-name ConversationHistory

# Clean up - delete the table
aws dynamodb delete-table --table-name ConversationHistory
